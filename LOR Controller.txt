   def brightnessLOR(self, b):
      # scale b from (0,255) to (240,1) which is the LOR intensity range
      brightness = 240 - int(b * 239 / 255.0)
      return(bytearray([brightness]))

   def channelLOR(self, nChannel):
      MaxChannel = 1 << 6
      if nChannel >= MaxChannel:
         lsb = (nChannel % MaxChannel) | 0xC0;
         msb = (nChannel // MaxChannel) | 0x80;
         return(bytearray([lsb, msb]))
      else:
         val = nChannel | 0x80;
         return(bytearray([val]))

   def getBrightness(self, b):
      # scale b from (0,255) to (240,1) which is the LOR intensity range
      brightness = 240 - int (b * 239 / 255.0)
      return(bytearray([ brightness ]))

   def getChannel(self, nChannel):
      MaxChannel = 1 << 6
      if nChannel >= MaxChannel:
         lsb = (nChannel % MaxChannel) | 0xC0;
         msb = (nChannel // MaxChannel) | 0x80;
         return(bytearray([ lsb, msb ]))
      else:
         val = nChannel | 0x80;
         return(bytearray([ val ]))

   def heartbeatTimeout(self):
      if self.bSerialPort:
         self.mutex.lock()
         try:
            self.serialPort.write(bytearray(b"\x00\xFF\x81\x56\x00"))
         except Exception as e:
            QMessageBox.warning(self, "Error", f"Serial port connection lost!\n{e}")
            self.bSerialPort = False
            self.serialPort.close()
            self.timer.stop()
         self.mutex.unlock()

   def nodeLOR(self, node):
      unitID = 0x01
      actionChannelBrightness = 0x03
      baseChannel = (node.channel - 1) * 3
      r, g, b, a = node.color.getRgb() if node.color is not None else \
           ( 0x00, 0x00, 0x00, 0xFF )
      r = int(r * self.masterBrightness)
      g = int(g * self.masterBrightness)
      b = int(b * self.masterBrightness)
      preString = bytearray([0x00, unitID, actionChannelBrightness])
      postString = bytearray(b"\x00\x00")
      ch1 = preString + self.brightnessLOR(r) + self.channelLOR(baseChannel) + postString
      ch2 = preString + self.brightnessLOR(g) + self.channelLOR(baseChannel+1) + postString
      ch3 = preString + self.brightnessLOR(b) + self.channelLOR(baseChannel+2) + postString
      return(ch1 + ch2 + ch3)


   def outputLOR(self):
      if self.bSerialPort:
         data = bytearray()
         self.mutex.lock()
         for node in self.modelNodes:
            data += self.nodeLOR(node)
         try:
            self.serialPort.write(data)
         except Exception as e:
            QMessageBox.warning(self, "Error", f"Serial port connection lost!\n{e}")
            self.bSerialPort = False
            self.serialPort.close()
            self.timer.stop()
         self.mutex.unlock()

   def outputToLights(self):
      bChecked = self.mainWindow.ui.actionOutput.isChecked()
      if bChecked:
         try:
            self.serialPort = serial.Serial(port="COM3", baudrate=500000, write_timeout=1.0)
            self.serialPort.set_buffer_size(rx_size=8192, tx_size=8192)  # Default are 4096 bytes
            self.timer.start(500)
            self.bSerialPort = True
         except Exception as e:
            QMessageBox.warning(self, "Error", f"Unable to connect to lights!\n{e}")
      else:
         self.bSerialPort = False
         self.serialPort.close()
         self.timer.stop()"
